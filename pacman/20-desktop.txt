# 20-desktop.txt — Hyprland 桌面环境 —— 装完这组就有可用图形界面
#
# 只收 Hyprland 这一系的桌面。整套 Plasma 会话(plasma-meta 那一坨)在 00-local.txt,
# 不入库 —— 它是本机的历史包袱,不该跟着仓库复制到新机器上。
# 下面的 KDE *应用*(dolphin/okular/gwenview…)留在这里:它们不依赖 Plasma 会话,
# 而且 mimeapps.list、vars.lua、windowrules.lua 直接点名了它们。
#
# 行尾注释写的是"这台机器为什么需要它",不是 pacman 的包描述 —— 新增包时照此写。
# [AUR] 标记的包必须用 paru/yay 安装,纯 pacman 装不了。

# ── Hyprland 核心 / 会话 / 锁屏 ─────────────────────────────────────────
hyprland                     # Wayland 合成器本体
uwsm                         # 通用 Wayland 会话管理(autostart/binds 里 uwsm-app 拉起应用)
xdg-desktop-portal-hyprland  # 截图/录屏/文件选择 portal 后端
xwayland-satellite           # 按需启动的 Xwayland
hypridle                     # 空闲管理(配合 hyprlock)
hyprlock                     # 锁屏
hyprpaper                    # 壁纸(备用,当前配置默认用 awww)
hyprpicker                   # 取色器(截图冻结画面)
awww                         # 壁纸守护(wallpaper.sh / autostart 的 awww-daemon 实际使用)
polkit-gnome                 # 图形化授权代理
sddm  # 登录/显示管理器(sddm.service enabled;原先只是 sddm-kcm 的依赖,险些被孤儿清理带走)

# ── 状态栏 / 启动器 / 通知 / 注销 ───────────────────────────────────────────
waybar       # 顶部状态栏
quickshell   # 桌面 shell:音量/亮度 OSD + 通知守护(替代 mako),见 config/quickshell/
vicinae-bin  # [AUR] 应用启动器(AUR,配置里默认用它);通知历史也走它的 dmenu 子命令

# ── 截图 / 录屏 / 剪贴板 ─────────────────────────────────────────────────
grim                 # 截图后端
slurp                # 区域选择
tensaku-bin          # [AUR] 截图标注(AUR,satty fork,支持滚动长截图);读 config/satty/
spectacle            # KDE 截图工具;主力是 hypr/scripts/screenshot.sh,这个作备用
gpu-screen-recorder  # 硬件加速录屏;RTX 4080 上走 NVENC,比 wl-screenrec 省 CPU
wl-screenrec-git     # 录屏(archlinuxcn / AUR);screenrecord.sh 默认用它
wl-clipboard         # wl-copy / wl-paste;截图脚本和 alias/git 都在用
xclip                # X11 剪贴板;config/zellij/config.kdl 的 copy_command 配的就是它
cliphist             # 剪贴板历史
wtype                # Wayland 下模拟键盘输入;注意它不触发 Hyprland keybind
zbar                 # zbarimg 二维码解码(screenshot.sh qr 模式)

# ── 密钥环 ───────────────────────────────────────────────────────────
gnome-keyring  # 密钥环 / SSH agent(autostart + env.lua 的 SSH_AUTH_SOCK)
seahorse       # 图形化管理 gnome-keyring 里的密钥
lssecret-git   # [AUR] 命令行列出 secret service 里的条目,排查 keyring 用

# ── KDE 应用(只是应用,不需要 Plasma 会话) ──────────────────────────────────
dolphin         # 文件管理器(vars.lua,SUPER+E)
gwenview        # 看图;mimeapps.list 里是图片的默认打开方式
okular          # 看 PDF;mimeapps.list 里是文档默认
filelight       # 图形化磁盘占用分析
isoimagewriter  # 把 ISO 写进 U 盘做启动盘
ark             # 压缩包 GUI。卸 plasma 后 zip 的默认关联掉给了 PrismLauncher,
                # tar/gz/rar/7z 直接无默认;装它零新增依赖(KF6 库 dolphin 已拉着)
kimageformats   # 给 Qt 应用补 AVIF/HEIF/JXL 图片格式(gwenview 打开这些格式靠它)
libheif         # HEIF/AVIF 的编解码库本体;kimageformats 和 gimp 都要它出图
breeze-gtk      # GTK 应用的 Breeze 主题(gtk-theme-name=Breeze 靠它;已从 plasma 依赖转显式)

# ── 远程桌面 (见 HANDOFF: wayvnc 走 frp+ssh 隧道) ─────────────────────────
wayvnc    # wlroots 的 VNC 服务端;只绑 127.0.0.1,跨公网走 frp+ssh 隧道
tigervnc  # VNC 客户端,连别的机器用

# ── 亮度 / 媒体键 ───────────────────────────────────────────────────────
ddcutil        # 外接显示器 DDC/CI 亮度;本机全外接,brightness.sh 只走这条路径
               # (笔记本要的 brightnessctl 在 00-local.example.txt 里)
playerctl      # MPRIS 播放控制;binds.lua 的播放/暂停/上下曲快捷键调它

# ── 音频 / 蓝牙前端 ─────────────────────────────────────────────────────
pipewire         # 音频/视频服务端(原先非显式,依赖链经 kwin,plasma 移除后有孤儿风险)
pipewire-pulse   # PulseAudio 兼容层
wireplumber      # pipewire 会话管理(wireplumber.service enabled,音量快捷键的 wpctl 来自它)
pipewire-audio   # ALSA 应用直接走 PipeWire:libasound_module_{pcm,ctl}_pipewire.so
                 # 加 /etc/alsa/conf.d/50-pipewire.conf 路由配置,都在这个包里。
                 # 取代了旧的 pulseaudio-alsa —— 那条路径要绕一层 Pulse API。
                 # 别跟 pipewire-alsa 搞混:那个包现在只剩 5 KiB,单独补一份
                 # 99-pipewire-default.conf 设 pcm.!default,实际路由不归它管
alsa-utils       # alsamixer/aplay/speaker-test;声音出问题时从最底层往上查
pavucontrol      # 图形音量面板
pamixer          # 音量控制(备用)
easyeffects      # PipeWire 音效处理(均衡器/降噪)
lsp-plugins-lv2  # easyeffects 用的 LV2 插件集
gst-plugins-good # GStreamer 常用解码器;GTK 应用内嵌播放靠它
blueman          # 蓝牙托盘
bluetui          # 终端里管蓝牙;blueman 的命令行替代

# ── 托盘 / 全局菜单兼容 ───────────────────────────────────────────────────
appmenu-gtk-module  # GTK 应用把菜单导出到 DBus,给全局菜单用
libdbusmenu-glib    # DBus 菜单协议库
libdbusmenu-gtk3    # 同上,GTK3 侧
libdbusmenu-qt5     # 同上,Qt5 侧
gtk2                # 遗留 GTK2 应用的运行时

# ── 输入法 (fcitx5) ──────────────────────────────────────────────────
fcitx5                 # 输入法框架
fcitx5-anthy           # fcitx5 的日文输入
fcitx5-chinese-addons  # 拼音/中文输入
fcitx5-configtool      # 图形配置
fcitx5-gtk             # GTK 应用集成
fcitx5-qt              # Qt 应用集成
fcitx5-material-color  # fcitx5 候选框主题
fcitx5-pinyin-zhwiki   # 中文维基词条做的拼音词库;打专有名词全靠它

# ── 光标 / 音效主题 ──────────────────────────────────────────────────────
adwaita-cursors         # 光标主题(gtk settings.ini + XCURSOR_THEME 都指向 Adwaita)
sound-theme-freedesktop # GTK 提示音(gtk-sound-theme-name=freedesktop)

# 字体拆到 25-fonts.txt 了(它是桌面的前置,不是应用,所以排在 20 和 30 之间)。

# ── 终端 / 缩略图 ─────────────────────────────────────────────────────
kitty              # 配置里默认终端
ffmpegthumbnailer  # dolphin 里显示视频缩略图(录屏文件多,没它只有一片空白图标)
