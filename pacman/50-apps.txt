# ── 代理 / VPN ──────────────────────────────────────────────────────

gost            # 通用隧道/端口转发,临时搭链路

frpc            # 内网穿透客户端;wayvnc 跨公网就走它
tailscale       # 组网 VPN;已装未启用
openvpn3        # [AUR] OpenVPN 3 客户端。**必须用干净 PATH 构建**:
                #   PATH="/usr/bin:$PATH" paru -S --rebuild openvpn3
                #   否则项目里的旧 protoc 会盖住系统版本导致编译失败,见 README 坑 12;
                #   abseil/protobuf 升级后还需重建,见坑 11
